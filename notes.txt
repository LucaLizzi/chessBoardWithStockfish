ARDUINO 
-> upload a code which can read and send serial commands

void setup() {
  Serial.begin(9600);
}

void loop() {
  if (Serial.available()) {
    String data = Serial.readStringUntil('\n');
    Serial.print("Received: ");
    Serial.println(data);
  }
}

#####################################################

PYTHON
-> write a command which can read and write serial commands

import serial
import time

# Open the serial port (adjust COM port or /dev/tty accordingly)
arduino = serial.Serial(port='COM3', baudrate=9600, timeout=1)
time.sleep(2)  # Give Arduino time to reset

# Send data
arduino.write(b'Hello Arduino\n')  # Always end with \n or matching delimiter

# Receive data
response = arduino.readline().decode().strip()
print("Arduino says:", response)

arduino.close()
